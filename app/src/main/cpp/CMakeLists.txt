cmake_minimum_required(VERSION 3.22)
project(xilon_llama LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
option(XILON_VULKAN "Build optional Vulkan inference backend" OFF)
set(GGML_VULKAN ${XILON_VULKAN} CACHE BOOL "" FORCE)
FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b12728 GIT_SHALLOW TRUE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
option(XILON_VLM "Build experimental multimodal projector support" OFF)
set(LLAMA_BUILD_MTMD ${XILON_VLM} CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
if(XILON_VULKAN AND DEFINED ENV{XILON_SPIRV_HEADERS_DIR})
 set(SPIRV-Headers_DIR "$ENV{XILON_SPIRV_HEADERS_DIR}" CACHE PATH "SPIR-V header package configuration" FORCE)
endif()
FetchContent_MakeAvailable(llama)
if(XILON_VULKAN)
 if(NOT DEFINED XILON_VULKAN_HEADERS_INCLUDE OR NOT EXISTS "${XILON_VULKAN_HEADERS_INCLUDE}/vulkan/vulkan.hpp")
  message(FATAL_ERROR "Vulkan-Hpp headers missing: set XILON_VULKAN_HEADERS_INCLUDE to a directory containing vulkan/vulkan.hpp")
 endif()
 if(NOT TARGET ggml-vulkan)
  message(FATAL_ERROR "Expected ggml-vulkan target is missing")
 endif()
 if(NOT EXISTS "${XILON_VULKAN_HEADERS_INCLUDE}/spirv/unified1/spirv.hpp")
  message(FATAL_ERROR "SPIR-V headers missing: expected spirv/unified1/spirv.hpp under XILON_VULKAN_HEADERS_INCLUDE")
 endif()
 target_include_directories(ggml-vulkan BEFORE PRIVATE "${XILON_VULKAN_HEADERS_INCLUDE}")
 # Use Vulkan-Hpp dynamic dispatch instead of linking Vulkan 1.1 entry points
 # that Android API 26 libvulkan.so does not export (e.g. vkGetPhysicalDeviceFeatures2).
 target_compile_definitions(ggml-vulkan PRIVATE VK_NO_PROTOTYPES)
endif()
add_library(xilon_llama SHARED bridge.cpp)
target_compile_features(xilon_llama PRIVATE cxx_std_17)
target_link_libraries(xilon_llama PRIVATE llama)

if(XILON_VLM)
 target_link_libraries(xilon_llama PRIVATE mtmd)
 target_compile_definitions(xilon_llama PRIVATE XILON_VLM=1)
endif()
